Bash Automations

UNIX Shell Scripting

This assignment consists of writing robust bash automation scripts to manage local directory directories, log system health parameters, and automate downloads.

Shell Script Database

#!/bin/bash
# a_Backup_Directory.sh
# Creates a backup directory and archives contents

SOURCE_DIR="$1"
BACKUP_DIR="$2"

if [ -z "$SOURCE_DIR" ] || [ -z "$BACKUP_DIR" ]; then
    echo "Usage: $0 [source_dir] [backup_dir]"
    exit 1
fi

mkdir -p "$BACKUP_DIR"
TIMESTAMP=$(date +%Y%m%d_%H%M%S)
tar -czf "$BACKUP_DIR/backup_$TIMESTAMP.tar.gz" "$SOURCE_DIR"

echo "Backup of $SOURCE_DIR completed successfully at $BACKUP_DIR/backup_$TIMESTAMP.tar.gz"
#!/bin/bash
# b_Usage_Log.sh
# Logs CPU and memory usage statistics

LOG_FILE="system_usage.log"

echo "=== Logging System Performance: $(date) ===" >> "$LOG_FILE"
echo "--- CPU Stats ---" >> "$LOG_FILE"
top -b -n 1 | head -n 10 >> "$LOG_FILE"
echo "--- Memory Stats ---" >> "$LOG_FILE"
free -h >> "$LOG_FILE"
echo "=========================================" >> "$LOG_FILE"

echo "Logged successfully to $LOG_FILE"
#!/bin/bash
# c_Download_Automation.sh
# Automatically sorts files in a download directory

DOWNLOADS_DIR="$HOME/Downloads"

mkdir -p "$DOWNLOADS_DIR/Images"
mkdir -p "$DOWNLOADS_DIR/Documents"
mkdir -p "$DOWNLOADS_DIR/Archives"

find "$DOWNLOADS_DIR" -maxdepth 1 -type f | while read file; do
    case "$file" in
        *.jpg|*.png|*.gif|*.webp) mv "$file" "$DOWNLOADS_DIR/Images/" ;;
        *.pdf|*.docx|*.txt|*.xlsx) mv "$file" "$DOWNLOADS_DIR/Documents/" ;;
        *.zip|*.tar.gz|*.rar) mv "$file" "$DOWNLOADS_DIR/Archives/" ;;
    esac
done

echo "Download directories sorted and organized."
